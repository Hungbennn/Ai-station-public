```python
"""D157: ngoại lệ đo G3.8 cho tram-kiem trên 24G90; khe thợ chờ G5 riêng.
Chỉ đọc đường dẫn executable từ libproc, không tin argv/comm/env. Lỗi hoặc đổi định danh đóng cổng.
"""
import ctypes
import os

from tram.kiem.bang_tien_trinh import chup_nhan, mot_nhan

_METADATA = "/System/Library/Frameworks/CoreServices.framework/Frameworks/Metadata.framework/Versions/A/Support/"
_CARBON = "/System/Library/Frameworks/CoreServices.framework/Versions/A/Frameworks/CarbonCore.framework/Versions/A/"
_NETFS = "/System/Library/Frameworks/NetFS.framework/Versions/A/XPCServices/"
APPLE_24G90 = frozenset({
    "/usr/sbin/distnoted", "/usr/libexec/trustd", "/usr/libexec/secd", "/usr/libexec/pkd",
    "/usr/libexec/lsd", "/usr/libexec/containermanagerd",
    _METADATA + "mdworker_shared", _METADATA + "mdbulkimport",
    _CARBON + "XPCServices/csnameddatad.xpc/Contents/MacOS/csnameddatad",
    _NETFS + "PlugInLibraryService.xpc/Contents/MacOS/PlugInLibraryService",
})


def build_macos():
    """Build của nhân đang chạy; không nhận build qua biến môi trường."""
    libc = ctypes.CDLL("/usr/lib/libSystem.B.dylib", use_errno=True)
    buf, n = ctypes.create_string_buffer(128), ctypes.c_size_t(128)
    if libc.sysctlbyname(b"kern.osversion", buf, ctypes.byref(n), None, 0) != 0:
        raise OSError(ctypes.get_errno(), "không đọc được build macOS — đo lại")
    return buf.value.decode("ascii", errors="strict")


def duong_nhan(pid):
    """proc_pidpath trả executable của kernel, không phải tên agent đặt cho argv[0]."""
    lib = ctypes.CDLL("/usr/lib/libproc.dylib", use_errno=True)
    lib.proc_pidpath.argtypes = [ctypes.c_int, ctypes.c_void_p, ctypes.c_uint32]
    lib.proc_pidpath.restype = ctypes.c_int
    buf = ctypes.create_string_buffer(4096)
    n = lib.proc_pidpath(pid, buf, len(buf))
    if n <= 0 or n >= len(buf) or not buf.value.startswith(b"/"):
        raise OSError(ctypes.get_errno(), "không đọc được executable của nhân")
    return os.fsdecode(buf.value)


def con_uid(uid, chup=chup_nhan, *, ten=None, ban=None, mot=mot_nhan, duong=duong_nhan):
    """Trả mọi tàn dư; chỉ principal đã đo mới có ngoại lệ. Không gửi tín hiệu."""
    mien = ten == "tram-kiem"
    if mien and (build_macos() if ban is None else ban) != "24G90":
        raise OSError("D157: build macOS chưa đo — đo lại G3 trước khi mở cổng")
    con = []
    for t in chup():
        if uid not in (t.uid, t.ruid):
            continue
        if not mien or t.ppid != 1 or t.uid != uid or t.ruid != uid or t.khoa is None:
            con.append(t)
            continue
        if mot(t.pid) != t:
            raise OSError("D157: tiến trình đổi trước khi đọc executable")
        path = duong(t.pid)
        if mot(t.pid) != t:
            raise OSError("D157: tiến trình đổi sau khi đọc executable")
        if path not in APPLE_24G90:
            con.append(t)
    return con
```
