# Lát 1b-2 — MCP memory cho CLI máy thợ (theo phiếu) và phiên tay chỉ đọc trên máy A — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use ai-skills:subagent-driven-development (recommended) or ai-skills:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Memory service (`tram nho chay`, 127.0.0.1:7710) phục vụ thêm MCP HTTP ở `/mcp` với đúng hai công cụ không có tham số phạm vi — `nho_lai(truy_van)` và `ghi_nho(noi_dung, nhan?)` — cho (a) CLI trên máy thợ, vùng cấp lấy từ phiếu đang giữ (`X-Tram-Giao-Id` + `X-Tram-Ma-Giao`, tra tram-api bằng `tram-noi-bo-token`, cache 30 giây) và (b) phiên tay của Đại ca trên máy A, chỉ đọc (`Authorization: Bearer <mã Đại ca>` + `X-Tram-Du-An`, trạm giữ sha256 của mã).

**Architecture:** Một cổng ASGI (`tram/nho/ung_dung.py`) đứng trước mọi thứ ở 7710: đường `/mcp` phải qua `xac_thuc` (`tram/nho/danh_tinh.py`) — lời gọi không có bằng chứng hợp lệ nhận 401/403/503 `{loi}` và không chạm SDK; hợp lệ thì `DanhTinh` (dự án, vùng cấp, được ghi hay không) đặt vào `scope["state"]` rồi chuyển cho máy chủ MCP của SDK `mcp` (`tram/nho/may_chu_mcp.py` — nơi DUY NHẤT import SDK, FastMCP streamable HTTP không phiên, trả JSON). Đường khác giữ nguyên HTTP nội bộ 1b-1 (`/nho-lai`, `/xoa`, Bearer `tram-nho-token`). Logic hai công cụ không phụ thuộc SDK (`tram/nho/cong_cu.py`): `nho_lai` gọi `nho_lai()` của 1b-1 với vùng cấp của `DanhTinh` và bọc mỗi ký ức trong `<du_lieu_khong_tin_cay>`; `ghi_nho` KHÔNG ghi kho trực tiếp mà ghi sự kiện nguồn `nho.agent_ghi` vào dự án của phiếu — vòng nạp 2 phút của 1b-1 biến nó thành ký ức qua đúng đường §6.2–§6.4 (không tách, luật cứng quyết nhánh, gộp có bảo vệ ký ức Đại ca, nhật ký thao tác idempotent). Tra phiếu: `tram/nho/tra_phieu.py` (urllib, timeout 2 giây, cache 30 giây, lỗi → từ chối). Mã Đại ca: `tram chu dat-ma-dai-ca` hỏi ẩn, lưu `ma_dai_ca_sha256` vào `~/.tram/cau-hinh.yaml` (`tram/nho/ma_dai_ca.py`); memory service đọc bản băm mỗi lời gọi.

**Tech Stack:** Python 3.12 (`uv`); **SDK `mcp==1.30.0`** (MIT — phụ thuộc mới duy nhất, spec §9.4 cho phép; G13); FastAPI 0.143.0 + uvicorn 0.54.0 (đã có); urllib stdlib; PyYAML (đã có); `pytest`.

**Spec:** [docs/specs/2026-10-13-phan-1b-memory.md](../specs/2026-10-13-phan-1b-memory.md) (ĐÃ DUYỆT 13/10, sửa lần 3 khớp mã 1b-1) — chỉ lát **1b-2**: §3 hàng 1b-2, §4 hai dòng cuối bảng "Ai cấp vùng", §2 M1/M2, §6.1 dòng "Agent tự ghi", §6.2 bước 2 (nhánh của `ghi_nho`), §6.4 ca "MCP `ghi_nho` trong lúc dọn phiên bản/sao lưu", §7 (đầu ra bọc thẻ), §9 sơ đồ, §9.2, §9.3, §9.4 (phụ thuộc `mcp`), §13 câu 8–10, §14 N2, §15 (phần 1b-2), §16.1 (các ca MCP). def doc_token_noi_bo(*, chay=subprocess.run) -> str | None:
    """Đọc Keychain; test thay `chay`, không bao giờ đọc Keychain thật."""
    kq = chay(["security", "find-generic-password", "-s", KEYCHAIN_NOI_BO, "-w"], capture_output=True, text=True)
    return (kq.stdout.strip() or None) if kq.returncode == 0 else None


def _dung_dang(d) -> bool:
    return (isinstance(d, dict) and all(k in d for k in TRUONG) and isinstance(d.get("du_an"), str) and isinstance(d.get("vai"), str)
            and isinstance(d.get("con_hieu_luc"), bool) and (d.get("ma_giao") is None or isinstance(d["ma_giao"], str)))

                 dong_ho: Callable[[], float] = time.monotonic, cache_giay: float = CACHE_GIAY):
        self._lay_token, self._url, self._mo, self._dong_ho = lay_token, url, mo, dong_ho
        self._cache_giay, self._khoa = cache_giay, threading.Lock()
        self._cache: OrderedDict[str, tuple[float, dict | None]] = OrderedDict()

    def __call__(self, giao_id: str) -> dict | None:
        if not isinstance(giao_id, str) or not GIAO_ID_RE.fullmatch(giao_id):
            raise ValueError("giao_id sai dạng")
        luc = self._dong_ho()
        with self._khoa:
            o = self._cache.get(giao_id)
            if o is not None and luc - o[0] < self._cache_giay:
                return o[1]
        kq = self._hoi(giao_id)
        with self._khoa:
            self._cache[giao_id] = (luc, kq)
            self._cache.move_to_end(giao_id)
            while len(self._cache) > TOI_DA_CACHE:
                self._cache.popitem(last=False)
        return kq

    def _hoi(self, giao_id: str) -> dict | None:
        token = self._lay_token()
        if not token:
            raise LoiTraPhieu("chưa có tram-noi-bo-token")
        yc = urllib.request.Request(f"{self._url}/api/noi-bo/phieu/{giao_id}",
                                    headers={"Authorization": f"Bearer {token}"})
        try:
            with self._mo(yc, timeout=THOI_GIAN_CHO) as r:
                d = json.loads(r.read(65536) or b"null")
        except urllib.error.HTTPError as e:
            if e.code == 404:
                return None
            raise LoiTraPhieu(f"tram-api trả {e.code}") from None
        except (urllib.error.URLError, OSError, ValueError, http.client.HTTPException) as e:
            raise LoiTraPhieu(f"không gọi được tram-api: {type(e).__name__}") from None
        if not _dung_dang(d):
            raise LoiTraPhieu("tram-api trả dữ liệu sai dạng")
        return {k: d[k] for k in TRUONG}
```

  vùng `[/du-an/X, /vai/<vai của phiếu>, /chung]`, được ghi (`ghi_nho` luôn vào `/du-an/X`).
- Phiên tay của Đại ca: `Authorization: Bearer <mã Đại ca>` + `X-Tram-Du-An: X` → so sha256 với bản băm trong

Header danh tính Tailscale không được xét (tiến trình cục bộ giả được — §9.3). Thông báo từ chối không chứa
`ma_giao` hay mã Đại ca."""
import hashlib
import hmac
import re
from collections.abc import Callable, Iterable
from dataclasses import dataclass, field

from tram.nho.pham_vi import TEN, goc_du_an

GIAO_ID_RE = re.compile(r"g-\d{8}-\d{4,}")
MA_GIAO_RE = re.compile(r"[A-Za-z0-9_-]{22}")
TEN_RE = re.compile(TEN)
VAI_PHIEN_TAY = "dieu-phoi"
H_GIAO, H_MA, H_AUTH, H_DU_AN = b"x-tram-giao-id", b"x-tram-ma-giao", b"authorization", b"x-tram-du-an"


class TuChoi(Exception):
    """`ma`: 401 bằng chứng thiếu/sai, 403 dự án không có, 503 không tra được phiếu — đều là từ chối (M2)."""

    def __init__(self, ly_do: str, ma: int = 401):
        super().__init__(ly_do)
        self.ma = ma


@dataclass(frozen=True)
class DanhTinh:
    du_an: str
    vung_cap: tuple[str, ...]
    duoc_ghi: bool
    giao_id: str | None = None
    vai: str | None = None
    ma_giao: str | None = field(default=None, repr=False)     # chỉ để che trong nội dung ghi_nho — không log


def _mot(headers: list[tuple[bytes, bytes]], ten: bytes) -> str | None:
    ds = [v for k, v in headers if k.lower() == ten]
    if len(ds) > 1:
        raise TuChoi("header lặp")
    return ds[0].decode("latin-1").strip() if ds else None


def xac_thuc(headers: Iterable[tuple[bytes, bytes]], *, tra: Callable[[str], dict | None] | None,
             bam_ma_dai_ca: Callable[[], str | None], co_du_an: Callable[[str], bool]) -> DanhTinh:
    """`headers`: header ASGI thô. `tra`: TraPhieu (None khi chưa có tram-noi-bo-token). Lỗi → TuChoi."""
    headers = list(headers)
    giao_id, ma_giao, auth, du_an = (_mot(headers, h) for h in (H_GIAO, H_MA, H_AUTH, H_DU_AN))
    if giao_id is not None or ma_giao is not None:
        if auth is not None or du_an is not None:
            raise TuChoi("trộn bằng chứng phiếu với mã Đại ca / X-Tram-Du-An")
        return _theo_phieu(giao_id, ma_giao, tra)
    if auth is not None or du_an is not None:
        return _phien_tay(auth, du_an, bam_ma_dai_ca, co_du_an)
    raise TuChoi("thiếu bằng chứng (phiếu hoặc mã Đại ca)")


def _theo_phieu(giao_id: str | None, ma_giao: str | None, tra) -> DanhTinh:
    if not giao_id or not ma_giao or not GIAO_ID_RE.fullmatch(giao_id) or not MA_GIAO_RE.fullmatch(ma_giao):
        raise TuChoi("thiếu hoặc sai dạng X-Tram-Giao-Id / X-Tram-Ma-Giao")
    if tra is None:
        raise TuChoi("memory service chưa có tram-noi-bo-token — không tra được phiếu", 503)
    try:
        p = tra(giao_id)
    except Exception as e:                                   # noqa: BLE001 — mọi lỗi tra phiếu: từ chối (fail closed)
        raise TuChoi(f"không tra được phiếu: {type(e).__name__}", 503) from None
    if p is None or p.get("con_hieu_luc") is not True or not isinstance(p.get("ma_giao"), str):
        raise TuChoi("phiếu không có hoặc đã hết hiệu lực")
    if not hmac.compare_digest(ma_giao.encode(), p["ma_giao"].encode()):
        raise TuChoi("ma_giao không khớp phiếu")
    du_an, vai = p.get("du_an"), p.get("vai")
    if not all(isinstance(x, str) and TEN_RE.fullmatch(x) for x in (du_an, vai)):
        raise TuChoi("phiếu mang dự án/vai sai dạng", 503)
    return DanhTinh(du_an, (goc_du_an(du_an), f"/vai/{vai}", "/chung"), True, giao_id, vai, ma_giao)


def _phien_tay(auth: str | None, du_an: str | None, bam_ma_dai_ca, co_du_an) -> DanhTinh:
    if not auth or not auth.startswith("Bearer ") or not du_an or not TEN_RE.fullmatch(du_an):
        raise TuChoi("phiên tay cần Authorization: Bearer <mã Đại ca> và X-Tram-Du-An: <dự án>")
    bam = bam_ma_dai_ca()
    if not bam:
        raise TuChoi("chưa đặt mã Đại ca (tram chu dat-ma-dai-ca)")
    gui = hashlib.sha256(auth.removeprefix("Bearer ").encode()).hexdigest()
    if not hmac.compare_digest(gui.encode(), bam.encode()):
        raise TuChoi("sai mã Đại ca")
    if not co_du_an(du_an):
        raise TuChoi(f"không có dự án {du_an}", 403)
    return DanhTinh(du_an, (goc_du_an(du_an), f"/vai/{VAI_PHIEN_TAY}", "/chung"), False)

biến môi trường, log hay màn hình: `tram chu dat-ma-dai-ca` hỏi ẩn (getpass) hai lần, chờ `dong-y`."""
import getpass
import hashlib
import os
import re
import sys
import tempfile
from pathlib import Path

import yaml

from tram.duong_dan import tram_home

KHOA = "ma_dai_ca_sha256"
MA_RE = re.compile(r"[A-Za-z0-9._~+/=-]{32,256}")       # an toàn trong header HTTP, đủ dài để không đoán được
BAM_RE = re.compile(r"[0-9a-f]{64}")
DONG_RE = re.compile(rf"^{KHOA}:.*$", re.M)


def tep_cau_hinh(duong_dan: Path | None = None) -> Path:
    return duong_dan or tram_home() / "cau-hinh.yaml"


def bam(ma: str) -> str:
    return hashlib.sha256(ma.encode()).hexdigest()


def _doc(van_ban: str) -> str | None:
    raw = yaml.safe_load(van_ban) or {}
    v = raw.get(KHOA) if isinstance(raw, dict) else None
    return v if isinstance(v, str) and BAM_RE.fullmatch(v) else None


def doc_bam(duong_dan: Path | None = None) -> str | None:
    """Bản băm đã đặt. Thiếu file, yaml hỏng, giá trị sai dạng → None: phiên tay bị từ chối (fail closed)."""
    try:
        return _doc(tep_cau_hinh(duong_dan).read_text(encoding="utf-8"))
    except (OSError, yaml.YAMLError):
        return None


def ghi_bam(gia_tri: str, duong_dan: Path | None = None) -> None:
    """Thay dòng `ma_dai_ca_sha256:` (hoặc thêm cuối file), giữ chú thích và khoá khác; kiểm trước, ghi nguyên tử."""
    if not BAM_RE.fullmatch(gia_tri):
        raise ValueError("bản băm phải là 64 ký tự hex thường")
    p = tep_cau_hinh(duong_dan)
    cu = p.read_text(encoding="utf-8") if p.exists() else ""
    dong = f'{KHOA}: "{gia_tri}"'
    if DONG_RE.search(cu):
        moi = DONG_RE.sub(dong, cu)
    else:
        moi = (cu if not cu or cu.endswith("\n") else cu + "\n") + dong + "\n"
    try:
        dung = _doc(moi) == gia_tri
    except yaml.YAMLError:
        dung = False
    if not dung:
        raise ValueError(f"{p} không phải bảng khoá → giá trị hợp lệ — sửa tay rồi chạy lại; không đổi gì")
    p.parent.mkdir(parents=True, exist_ok=True)
    fd, tam = tempfile.mkstemp(dir=p.parent, prefix=".cau-hinh-")
    with os.fdopen(fd, "w", encoding="utf-8") as f:
        f.write(moi)
    os.replace(tam, p)


def dat_ma_dai_ca(args, *, doc_an=None, hoi=None) -> int:
    doc_an, hoi, p = doc_an or getpass.getpass, hoi or input, tep_cau_hinh()
    print(f"Sẽ ghi sha256 của mã Đại ca vào {p} (khoá {KHOA}; thay bản cũ nếu có). Mã không được lưu, không in.\n"
          "Mã: chuỗi ngẫu nhiên ≥ 32 ký tự Đại ca tạo trên máy A (ví dụ `openssl rand -hex 32`), đúng mã đặt vào "
          "TRAM_DAI_CA_MA của cấu hình MCP trên máy A.\nDán trong Terminal.app — khung terminal của app Claude làm "
          "hỏng chuỗi dán vào ô ẩn.")
    if not args.dong_y and hoi("Gõ dong-y để tiếp tục: ").strip() != "dong-y":
        print("Không đổi gì.")
        return 1
    ma = doc_an("Mã Đại ca (ẩn): ")
    if not MA_RE.fullmatch(ma):
        print("Lỗi: mã phải 32–256 ký tự trong [A-Za-z0-9._~+/=-] — không đổi gì.", file=sys.stderr)
        return 1
    if doc_an("Nhập lại (ẩn): ") != ma:
        print("Lỗi: hai lần nhập không khớp — không đổi gì.", file=sys.stderr)
        return 1
    try:
        ghi_bam(bam(ma), p)
    except ValueError as e:
        print(f"Lỗi: {e}", file=sys.stderr)
        return 1
    print(f"Đã lưu bản băm vào {p}. Memory service dùng ngay từ lời gọi kế tiếp, không cần khởi động lại.")
    return 0


def them_lenh_ma_dai_ca(chu) -> None:
    d = chu.add_parser("dat-ma-dai-ca", help="Đặt mã Đại ca cho phiên tay MCP (lưu sha256, hỏi ẩn)")
    d.add_argument("--dong-y", action="store_true", help="Bỏ bước gõ dong-y (vẫn hỏi mã ẩn)")
    d.set_defaults(ham=dat_ma_dai_ca)
```

import json

from tram.nho.pham_vi import goc_du_an

TEN_MAY_CHU = "tram-nho"


def cau_hinh_tho() -> dict:
    """CLI máy thợ: URL theo máy (`TRAM_NHO_URL`), bằng chứng là phiếu đang giữ (`TRAM_GIAO_ID`, `TRAM_MA_GIAO`)."""
    return {"type": "http", "url": "${TRAM_NHO_URL}",
            "headers": {"X-Tram-Giao-Id": "${TRAM_GIAO_ID}", "X-Tram-Ma-Giao": "${TRAM_MA_GIAO}"}}

    """Phiên tay của Đại ca trên máy A: mã Đại ca từ biến `TRAM_DAI_CA_MA` (không nằm trong file), dự án tĩnh."""

        raise ValueError("url phải dạng https://<máy B>.<tailnet>.ts.net/nho/mcp")
    return {"type": "http", "url": url, "headers": {"Authorization": "Bearer ${TRAM_DAI_CA_MA}", "X-Tram-Du-An": du_an}}


def mcp_json(may_chu: dict) -> str:
    return json.dumps({"mcpServers": {TEN_MAY_CHU: may_chu}}, ensure_ascii=False, indent=2) + "\n"

            if callable(v):
                v()
            else:
                kq.append(await _http(app, *v))
    finally:
        await vao.put({"type": "lifespan.shutdown"})
        await asyncio.wait_for(ra.get(), 10)
        await asyncio.wait_for(chay, 10)
    return kq
